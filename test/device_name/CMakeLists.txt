cmake_minimum_required(VERSION 3.16)
project(device_name_host_tests CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
enable_testing()
get_filename_component(REPO_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
add_executable(DeviceNameTest DeviceNameTest.cpp "${REPO_ROOT}/src/network/DeviceName.cpp")
target_include_directories(DeviceNameTest PRIVATE "${REPO_ROOT}/src/network")
target_compile_options(DeviceNameTest PRIVATE -Wall -Wextra -pedantic)
add_test(NAME DeviceName COMMAND DeviceNameTest)
