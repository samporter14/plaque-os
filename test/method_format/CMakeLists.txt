cmake_minimum_required(VERSION 3.16)
project(method_format_host_tests CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
enable_testing()
get_filename_component(REPO_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
add_executable(MethodFormatTest MethodFormatTest.cpp "${REPO_ROOT}/src/ambient/MethodFormat.cpp")
target_include_directories(MethodFormatTest PRIVATE "${REPO_ROOT}/src/ambient" "${REPO_ROOT}/test/method_support")
target_compile_options(MethodFormatTest PRIVATE -Wall -Wextra -pedantic)
add_test(NAME MethodFormat COMMAND MethodFormatTest)
